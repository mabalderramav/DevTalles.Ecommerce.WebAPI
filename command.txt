# Commands utility to manage Docker containers with docker-compose:
# Start containers in the background:
docker-compose up -d
# Stop and remove containers (does not remove volumes):
docker-compose down
# Build the project using the .NET CLI:
dotnet build
# Install the Entity Framework Core command-line tools globally:
dotnet tool install --global dotnet-ef
# Check the installed version of the Entity Framework Core tools:
dotnet ef --version
# Add the Entity Framework Core Design package to your project:
dotnet add package Microsoft.EntityFrameworkCore.Design --version 9.0.14
# Add the AutoMapper package to your project:
dotnet add package AutoMapper --version 16.1.1
# Add a new migration named "InitialMigration" to your project:
# Add the BCrypt.Net-Next package to your project for password hashing:
dotnet add package BCrypt.Net-Next --version 4.0.3

# Add packages for JWT authentication:
dotnet add package System.IdentityModel.Tokens.Jwt --version 8.14.0
dotnet add package Microsoft.IdentityModel.Tokens --version 8.14.0
## The Microsoft.IdentityModel.JsonWebTokens package provides functionality for creating and validating JSON Web Tokens (JWTs) in .NET applications. It is part of the Microsoft IdentityModel library, which offers various tools and components for handling authentication and authorization in .NET applications.
## this is optional, but it can be useful for working with JWTs in a more structured way, especially if you need to create or validate JWTs in your application:
dotnet add package Microsoft.IdentityModel.JsonWebTokens --version 8.14.0

# Add packages for versioning:
dotnet add package Asp.Versioning.Mvc.ApiExplorer --version 8.1.1
dotnet add package Asp.Versioning.Mvc --version 8.1.1

dotnet ef migrations add InitialMigration
# Remove the latest migration from your project:
dotnet ef migrations remove 
# Update the database to apply the latest migrations:
dotnet ef database update
# Connection string for SQL Server:
## Ubuntu/WSL, use host alias:
Server=host.docker.internal,1433;Database=ApiEcommerceNET8;User ID=SA;Password=$Aldo123$5;TrustServerCertificate=true;MultipleActiveResultSets=true
## Windows, use localhost:
Server=localhost,1433;Database=ApiEcommerceNET8;User ID=SA;Password=$Aldo123$5;TrustServerCertificate=true;MultipleActiveResultSets=true
# Update the database using a specific connection string: Ubuntu/WSL
dotnet ef database update --connection 'Server=host.docker.internal,1433;Database=ApiEcommerceNET8;User ID=SA;Password=$Aldo123$5;TrustServerCertificate=true;MultipleActiveResultSets=true;Encrypt=true'

# Docker image build and push to Docker Hub:
# (1) Login to Docker Hub
# docker login

# (2) Build image from Dockerfile
# docker build -t devtalles-ecommerce-webapi:latest .

# (3) Optional: add version tag
# docker tag <DOCKERHUB_USERNAME>/devtalles-ecommerce-webapi:latest <DOCKERHUB_USERNAME>/devtalles-ecommerce-webapi:v1.0.0
docker tag devtalles-ecommerce-webapi:latest mabalderramav/devtalles-ecommerce-webapi:v1.0.2

# (4) Push tags
# docker push <DOCKERHUB_USERNAME>/devtalles-ecommerce-webapi:latest
# docker push <DOCKERHUB_USERNAME>/devtalles-ecommerce-webapi:v1.0.0
docker push mabalderramav/devtalles-ecommerce-webapi:v1.0.2

# Docker compose commands for this project (uses .env automatically):
# Build and start services in background
# docker compose up --build -d

# Start services (without rebuild)
# docker compose up -d

# Stream logs
# docker compose logs -f

# Stop and remove containers/network
# docker compose down

# Stop and remove containers/network + anonymous volumes
# docker compose down -v

# Validate docker-compose.yaml after env substitution
# docker compose -f docker-compose.yaml config
